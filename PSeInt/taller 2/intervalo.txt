				Si numero >= 70 Y numero <= 120 Entonces

				Sino
					Si numero >= 121 Y numero <= 149 Entonces
						Escribir "El numero se encuentra entre 121 y 149"
					Sino
						Si numero >= 150 Y numero <= 300 Entonces
							Escribir "El numero se encuentra entre 150 y 300"
						Sino
							Si numero > 300 Entonces
								Escribir "El numero es mayor de 300"
							Sino

							FinSi
						FinSi

			FinSi
		FinSi
