Algoritmo Y_y_X
	Definir x, y1 Como Real
	
	Escribir "Ingrese el valor de X:"
	Leer x
	
	Si x < -4 Entonces
		y1 <- 3.15 * x + 5.25
	Sino
		Si x >= -4 Y x < 4 Entonces
			y1 <- x^2 - 16
		Sino
			Si x >= 4 Y x < 35 Entonces
				y1 <- 4.12 * x - 5
			Sino
				y1 <- x^5 - 7 * x
			FinSi
		FinSi
	FinSi
	
	Escribir "El valor de Y es: ", y1
FinAlgoritmo
